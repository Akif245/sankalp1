    let currentSpeech = null;
    let isSpeaking = false;

    function stopSpeech() {
      if (isSpeaking) {
        window.speechSynthesis.cancel();
        isSpeaking = false;
        document.getElementById("speechStatus").textContent = "Stopped";
        setTimeout(() => {
          document.getElementById("speechStatus").textContent = "";
        }, 2000);
      }
    }

    function speakText(text) {
      stopSpeech();
      const cleanText = text.replace(/[#*\[\]]/g, '');
      currentSpeech = new SpeechSynthesisUtterance(cleanText);
      currentSpeech.onstart = () => {
        isSpeaking = true;
        document.getElementById("speechStatus").textContent = "Speaking...";
      };
      currentSpeech.onend = () => {
        isSpeaking = false;
        document.getElementById("speechStatus").textContent = "Finished";
        setTimeout(() => {
          document.getElementById("speechStatus").textContent = "";
        }, 2000);
      };
      currentSpeech.onerror = () => {
        isSpeaking = false;
        document.getElementById("speechStatus").textContent = "Error";
      };
      window.speechSynthesis.speak(currentSpeech);
    }

    document.addEventListener("DOMContentLoaded", () => {
      const generateBtn = document.getElementById("generateBtn");
      const topicInput = document.getElementById("topic");
      const resultDiv = document.getElementById("result");
      const loadingDiv = document.querySelector('.loading');
      const askBtn = document.getElementById("askBtn");
      const questionInput = document.getElementById("question");
      const professorResponse = document.getElementById("professor-response");
      const speechControls = document.getElementById("speechControls");
      const speakBtn = document.getElementById("speakBtn");
      const stopBtn = document.getElementById("stopBtn");

      speakBtn.addEventListener("click", () => {
        const responseText = professorResponse.textContent;
        if (responseText && responseText !== "Thinking..." && !responseText.includes("Error")) {
          speakText(responseText);
        }
      });

      stopBtn.addEventListener("click", stopSpeech);

      generateBtn.addEventListener("click", async () => {
        const topic = topicInput.value.trim();
        if (!topic) {
          resultDiv.innerHTML = `<div class="error">⚠️ Please enter a topic</div>`;
          return;
        }

        generateBtn.disabled = true;
        loadingDiv.style.display = 'block';
        resultDiv.style.display = 'none';

        try {
          const res = await fetch("/generate_course", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ topic })
          });

          const data = await res.json();
          if (data.course) {
            resultDiv.innerHTML = `<h3 class="text-primary">Your ${topic} Course</h3>${marked.parse(data.course)}`;
          } else {
            resultDiv.innerHTML = `<div class="error">${data.error}</div>`;
          }
        } catch (err) {
          resultDiv.innerHTML = `<div class="error">Failed to connect to backend.</div>`;
        } finally {
          loadingDiv.style.display = 'none';
          resultDiv.style.display = 'block';
          generateBtn.disabled = false;
        }
      });

      topicInput.addEventListener("keypress", e => {
        if (e.key === "Enter") generateBtn.click();

        professorResponse.innerHTML = "<div class='loading-spinner'></div><p class='text-center'>Professor is thinking...</p>";
        speechControls.style.display = 'none';

        try {
          const res = await fetch("/ask_professor", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ question })
          });

          const data = await res.json();

            professorResponse.innerHTML = marked.parse(data.answer);
            speechControls.style.display = 'flex';
          } else {
            professorResponse.innerHTML = `<div class="error">${data.error}</div>`;
          }

        }

      });

          progressResult.innerHTML = `<div class="error">Failed to load progress</div>`;
        } finally {
          checkProgressBtn.disabled = false;
        }

def call_ai_api(prompt, system_message=None):
    """Call AI API with the given prompt (using OpenRouter)"""
    # Check if API key is available
    if not OPENROUTER_API_KEY:
        return "AI service is currently unavailable. Please check your API configuration."
    
    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY}",
        "Content-Type": "application/json",
        "HTTP-Referer": "http://localhost:5000",
        "X-Title": "AI Course Generator"
    }
    
    messages = []
    if system_message:
        messages.append({"role": "system", "content": system_message})
    messages.append({"role": "user", "content": prompt})
    
    payload = {
        "model": "deepseek/deepseek-chat",
        "messages": messages,
        "temperature": 0.7,
        "max_tokens": 2048
    }

@app.route("/evaluate_assessment", methods=["POST"])
def evaluate_assessment():
    data = request.get_json()
    module_name = data.get("module", "").strip()
    student_id = data.get("student_id", "anonymous")

    if not module_name:
        return jsonify({"error": "Module name required"}), 400

    try:
        if (student_id not in student_assessments or 
            'assessments' not in student_assessments[student_id] or 
            module_name not in student_assessments[student_id]['assessments'] or
            not student_assessments[student_id]['assessments'][module_name]['submission']):
            
            return jsonify({"error": "Submission not found"}), 404

        submission = student_assessments[student_id]['assessments'][module_name]['submission']
        assessment = student_assessments[student_id]['assessments'][module_name]['assessment']
        
        prompt = f"""
        As an expert professor, evaluate this student's assessment submission.
        
        ASSESSMENT BRIEF:
        {assessment}
        
        STUDENT'S SUBMISSION:
        {submission}
        
        Provide:
        1. Overall score (0-100)
        2. Strengths and what was done well
        3. Areas for improvement
        4. Specific feedback and suggestions
        5. Encouraging closing remarks
        
        Be constructive, educational, and supportive.
        """
        system_message = "You are a supportive professor evaluating student work. Provide constructive feedback that helps students learn and improve."
        
        feedback = call_ai_api(prompt, system_message)
        
        # Store feedback
        student_assessments[student_id]['assessments'][module_name]['feedback'] = feedback
        student_assessments[student_id]['assessments'][module_name]['status'] = "evaluated"
        
        # Update progress
        if 'progress' in student_assessments[student_id]:
            if module_name not in student_assessments[student_id]['progress']['completed']:
                student_assessments[student_id]['progress']['completed'].append(module_name)
        
        return jsonify({"feedback": feedback.strip(), "status": "evaluated"})
    except Exception as e:
        return jsonify({"error": str(e)}), 500
